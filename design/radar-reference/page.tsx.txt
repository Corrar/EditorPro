async function apiGet(endpoint: string, query?: Record<string, string>): Promise<any[]> {
  const qs = query ? '?' + new URLSearchParams(query).toString() : '';
  const r = await fetch(`/api/n8n/${endpoint}${qs}`, { cache: 'no-store' });
  if (r.status === 401) { window.location.href = '/login'; return []; }
  try { const d = await r.json(); return Array.isArray(d) ? d : []; } catch { return []; }
}
async function apiPost(endpoint: string, body: Record<string, unknown>): Promise<any> {
  const r = await fetch(`/api/n8n/${endpoint}`, { method: 'POST', headers: { 'content-type': 'application/json' }, body: JSON.stringify(body) });
  if (r.status === 401) { window.location.href = '/login'; return {}; }
  try { return await r.json(); } catch { return {}; }
}
const digits = (s: any) => String(s ?? '').replace(/[^0-9]/g, '');
// classifica o número coletado no Places: WhatsApp confirmado (a empresa divulga
// link wa.me no Google), celular (quase sempre atende WhatsApp) ou telefone fixo.
function foneTipo(it: any): { key: 'whats' | 'cel' | 'fixo'; label: string; full: string } | null {
  let d = digits(it?.telefone);
  if (!d) return null;
  if (d.length >= 12 && d.startsWith('55')) d = d.slice(2); // remove o +55
  if (/wa\.me\/|api\.whatsapp\.com|whatsapp\.com\//i.test(String(it?.site || '')))
    return { key: 'whats', label: 'WhatsApp', full: 'WhatsApp confirmado — a empresa divulga o link do WhatsApp no Google' };
  if (/^0[3589]00/.test(d))
    return { key: 'fixo', label: 'Fixo (0800)', full: 'Número de atendimento (0800/0300) — sem WhatsApp, melhor ligar' };
  const local = d.length >= 10 ? d.slice(2) : d; // remove o DDD
  if (/^[6-9]/.test(local))
    return { key: 'cel', label: 'Celular', full: 'Número de celular — quase sempre atende WhatsApp' };
  return { key: 'fixo', label: 'Fixo', full: 'Telefone fixo — dificilmente tem WhatsApp, melhor ligar' };
}
function FoneBadge({ it }: any) {
  const t = foneTipo(it);
  if (!t) return null;
  return <span className={`rl-badge tel ${t.key}`} title={t.full}><Icon n={t.key === 'whats' ? 'whatsapp' : t.key === 'cel' ? 'cel' : 'tel'} s={11} />{t.label}</span>;
}

function segmentOf(categoria?: string, servico?: string) {
  const s = `${categoria || ''} ${servico || ''}`.toLowerCase().normalize('NFD').replace(/[̀-ͯ]/g, '');
  if (/(fabric|metal|indust|distribuidor|atacad|usina|quimic)/.test(s)) return 'industria';
  if (/(transport|logist|entrega|frete|mudanc)/.test(s)) return 'transporte';
  if (/(loja|material|movei|movel|autopec|veicul|concession|magazine|comercio|varejo|mercado|deposito|conveni)/.test(s)) return 'comercio';
  return 'geral';
}

// sem SMTP — abre o Gmail/app de e-mail do Bruno). Corpo = mensagem de abordagem
// (a mesma do WhatsApp) + fecho próprio de e-mail com opt-out educado (LGPD).
function emailHref(it: any, corpo?: string) {
  const nome = it.nome || 'sua empresa';
  const oferta = it.servico_recomendado || 'presença digital';
  const assunto = `Uma ideia para a ${nome} aparecer mais no Google`;
  const base = (corpo || it.mensagem_abordagem || msgFor(nome, oferta, it.categoria)).trim();
  const body = `${base}\n\nSe preferir, é só responder este e-mail que eu te mostro os exemplos.\n(Se não quiser mais receber contato, é só me avisar por aqui.)\n\nAbraço,\nBruno`;
  return `mailto:${encodeURIComponent(String(it.email || '').trim())}?subject=${encodeURIComponent(assunto)}&body=${encodeURIComponent(body)}`;
}

    </div>
  );
}

/* -------------------- Análise de conversão -------------------- */
function AnaliseView({ a }: any) {
  if (!a) return <div className="rl-grid"><div className="rl-skel" /><div className="rl-skel" /><div className="rl-skel" /></div>;
  const f = a.funil || {};
  const num = (k: string) => Number(f[k]) || 0;
  const stages = [
    { k: 'qualificado', label: 'Qualificados', color: 'var(--green)' },
    { k: 'aprovado', label: 'Aprovados', color: '#7FBE9D' },
    { k: 'em_contato', label: 'Em contato', color: 'var(--amberInk)' },
    { k: 'negociacao', label: 'Em negociação', color: '#5B8DEF' },
    { k: 'fechado', label: 'Fechados', color: 'var(--green)' },
    { k: 'descartado', label: 'Descartados', color: 'var(--red)' },
  ];
  const maxFunil = Math.max(1, ...stages.map((s) => num(s.k)));
  const cidades: any[] = a.por_cidade || [];
  const nichos: any[] = a.por_nicho || [];
  const maxCid = Math.max(1, ...cidades.map((c) => Number(c.total) || 0));
  const maxNic = Math.max(1, ...nichos.map((n) => Number(n.total) || 0));
  const trabalhados = num('aprovado') + num('em_contato');
  const baseDecidida = trabalhados + num('descartado');
  const taxaAprov = baseDecidida > 0 ? Math.round((trabalhados / baseDecidida) * 100) : 0;
  return (
    <div className="rl-dash">
      <div className="rl-card" style={{ gridColumn: '1 / -1' }}>
        <div className="rl-title">Funil de conversão</div>
        <div className="rl-anfunil">
          {stages.map((s) => { const v = num(s.k); return (
            <div className="rl-anrow" key={s.k}>
              <span className="lbl">{s.label}</span>
              <div className="track"><i style={{ width: (v / maxFunil * 100) + '%', background: s.color }} /></div>
              <b className="mono">{v}</b>
            </div>
          ); })}
        </div>
        <div className="rl-anmetrics">
          <div className="rl-antile"><span className="mono big">{taxaAprov}%</span><span>aproveitamento (trabalhados ÷ decididos)</span></div>
          <div className="rl-antile"><span className="mono big">{trabalhados}</span><span>em trabalho</span></div>
          <div className="rl-antile"><span className="mono big">{num('fechado')}</span><span>fechados</span></div>
        </div>
      </div>

      <div className="rl-card">
        <div className="rl-title">Por cidade</div>
        {cidades.length === 0 ? <div className="rl-meta" style={{ marginTop: 6 }}>Sem dados ainda.</div>
          : <div className="rl-anfunil">{cidades.map((c) => (
            <div className="rl-anrow" key={c.cidade || '—'}><span className="lbl">{c.cidade || '—'}</span><div className="track"><i style={{ width: (Number(c.total) / maxCid * 100) + '%' }} /></div><b className="mono">{c.total}</b></div>
          ))}</div>}
      </div>

      <div className="rl-card">
        <div className="rl-title">Top nichos</div>
        {nichos.length === 0 ? <div className="rl-meta" style={{ marginTop: 6 }}>Sem dados ainda.</div>
          : <div className="rl-anfunil">{nichos.map((n, i) => (
            <div className="rl-anrow" key={(n.categoria || '') + i}><span className="lbl" title={n.categoria}>{n.categoria || '—'}</span><div className="track"><i style={{ width: (Number(n.total) / maxNic * 100) + '%' }} /></div><b className="mono">{n.total}</b></div>
          ))}</div>}
      </div>
    </div>
  );
}

/* -------------------- Kanban -------------------- */
function KanbanBoard({ data, loading, over, setOver, setDrag, onDrop, onOpen }: any) {
  const items = (col: string) => data[col] || [];
  const idOf = (col: string, it: any) => col === 'fila' || col === 'negociacao' ? it.lead_id : col === 'contratos' ? it.contrato_id : it.place_id;
  const titleOf = (col: string, it: any) => it.nome || it.razao_social || '—';
  const subOf = (col: string, it: any) => col === 'prospeccao' ? `score ${it.score_aderencia} · ${it.cidade}` : col === 'abordagens' ? `${it.cidade} · ${it.estado}` : it.servico_recomendado || it.segmento || '';
  return (
    <div className="rl-kwrap">
      {KCOLS.map((c) => (
        <div key={c.key} className={`rl-kcol${over === c.key ? ' over' : ''}`}
          onDragOver={(e) => { e.preventDefault(); if (over !== c.key) setOver(c.key); }}
          onDragLeave={() => setOver((o: string) => (o === c.key ? null : o))}
          onDrop={() => onDrop(c.key)}>
          <div className="rl-khead"><b>{c.label}</b><span className="mono">{loading[c.key] ? '·' : items(c.key).length}</span></div>
          {items(c.key).map((it: any) => (
            <div className="rl-kcard" key={idOf(c.key, it)} draggable onDragStart={() => setDrag({ col: c.key, item: it })}
              onClick={() => (c.key === 'prospeccao' || c.key === 'abordagens') && onOpen('prospeccao', it)}>
              <div className="t">{titleOf(c.key, it)}</div>
              <div className="s mono">{subOf(c.key, it)}</div>
            </div>
          ))}
          {items(c.key).length === 0 && !loading[c.key] && <div className="rl-kempty">vazio</div>}
        </div>
      ))}
    </div>
  );
}

/* -------------------- Detail modal (edição inline) -------------------- */
function DetailModal({ d, onClose, onSave }: any) {
  const it = d.item;
  const fields = [
    { k: 'nome', label: 'Nome' }, { k: 'servico_recomendado', label: 'Oferta sugerida' },
    { k: 'faixa_ticket', label: 'Faixa de ticket' }, { k: 'telefone', label: 'Telefone' },
    { k: 'email', label: 'E-mail' },
    { k: 'cidade', label: 'Cidade' }, { k: 'resumo', label: 'Resumo' },
  ];
  const init = () => { const o: Record<string, string> = {}; fields.forEach((f) => (o[f.k] = it[f.k] || '')); return o; };
  const [vals, setVals] = useState<Record<string, string>>(init);
  const [base, setBase] = useState<Record<string, string>>(init); // baseline salvo — evita re-POST no blur
  const save = (k: string) => { if (vals[k] !== (base[k] ?? '')) { onSave(it.place_id, k, vals[k]); setBase((b) => ({ ...b, [k]: vals[k] })); } };
  useEffect(() => { const onKey = (e: KeyboardEvent) => { if (e.key === 'Escape') onClose(); }; document.addEventListener('keydown', onKey); return () => document.removeEventListener('keydown', onKey); }, [onClose]);
  return (
    <div className="rl-modal-bg" onClick={onClose}>
      <div className="rl-modal" onClick={(e) => e.stopPropagation()} role="dialog" aria-modal="true">
        <div className="rl-modal-top"><div className="rl-title" style={{ fontSize: 18 }}>{vals.nome || it.nome}</div><button className="rl-iconbtn" onClick={onClose} aria-label="Fechar"><Icon n="x" s={16} /></button></div>
        <div className="rl-meta">Clique num valor para editar. {it.score_aderencia ? <>Score <b className="mono" style={{ color: 'var(--ink)' }}>{it.score_aderencia}</b></> : null}</div>
        {fields.map((f) => (
          <div className="rl-field" key={f.k}>
            <div className="rl-label">{f.label}</div>
            {f.k === 'resumo'
              ? <textarea className="rl-textarea" style={{ minHeight: 70 }} value={vals[f.k]} onChange={(e) => setVals({ ...vals, [f.k]: e.target.value })} onBlur={() => save(f.k)} />
              : <input className={`rl-input${f.k === 'telefone' || f.k === 'faixa_ticket' || f.k === 'email' ? ' mono' : ''}`} value={vals[f.k]} onChange={(e) => setVals({ ...vals, [f.k]: e.target.value })} onBlur={() => save(f.k)} />}
          </div>
        ))}
        <div className="rl-foot"><button className="rl-btn primary" onClick={onClose}>Concluir</button></div>
      </div>
    </div>
  );
}

/* -------------------- Assistente de resposta (IA) -------------------- */
function AssistenteModal({ it, onClose }: any) {
  const [cliente, setCliente] = useState('');
  const [resp, setResp] = useState('');
  const [loading, setLoading] = useState(false);
  const [copiado, setCopiado] = useState(false);
  const fone = digits(it.telefone);
  useEffect(() => { const onKey = (e: KeyboardEvent) => { if (e.key === 'Escape') onClose(); }; document.addEventListener('keydown', onKey); return () => document.removeEventListener('keydown', onKey); }, [onClose]);
  async function sugerir() {
    if (!cliente.trim()) return;

/* -------------------- Paginação -------------------- */
function Pager({ page, total, onPage }: { page: number; total: number; onPage: (p: number) => void }) {
  if (total <= 1) return null;
  const nums: (number | string)[] = [];
  const win = 2;
  for (let i = 1; i <= total; i++) {
    if (i === 1 || i === total || (i >= page - win && i <= page + win)) nums.push(i);
    else if (nums[nums.length - 1] !== '…') nums.push('…');
  }
  return (
    <div className="rl-pager">
      <button className="rl-pagebtn" disabled={page <= 1} onClick={() => onPage(page - 1)}>← Anterior</button>
      <div className="rl-pagenums">
        {nums.map((n, i) => n === '…'
          ? <span key={`e${i}`} className="rl-pagedots">…</span>
          : <button key={n} className={`rl-pagenum${n === page ? ' on' : ''}`} onClick={() => onPage(n as number)}>{n}</button>)}
      </div>
      <button className="rl-pagebtn" disabled={page >= total} onClick={() => onPage(page + 1)}>Próxima →</button>
    </div>
  );
}

  const prev = it.n_avaliacoes_anterior;

// horario_semana (Google Places): aberto agora (hora do navegador) + linha de hoje
function horarioInfo(it: any): { aberto: boolean | null; hoje: string | null } {
  let h = it?.horario_semana;
  if (!h) return { aberto: null, hoje: null };
  if (typeof h === 'string') { try { h = JSON.parse(h); } catch { return { aberto: null, hoje: null }; } }
  const now = new Date();
  const desc = Array.isArray(h.weekdayDescriptions) ? h.weekdayDescriptions[(now.getDay() + 6) % 7] || null : null;
  const per = Array.isArray(h.periods) ? h.periods : null;
  if (!per || !per.length) return { aberto: null, hoje: desc };
  const SEM = 7 * 1440;
  const nowAbs = now.getDay() * 1440 + now.getHours() * 60 + now.getMinutes();
  let aberto = false;
  for (const p of per) {
    const o = p?.open; if (!o) continue;
    const a = (o.day ?? 0) * 1440 + (o.hour || 0) * 60 + (o.minute || 0);
    let f = p?.close ? (p.close.day ?? 0) * 1440 + (p.close.hour || 0) * 60 + (p.close.minute || 0) : a + SEM;
    if (f <= a) f += SEM; // período vira a madrugada / 24h
    if ((nowAbs >= a && nowAbs < f) || (nowAbs + SEM >= a && nowAbs + SEM < f)) { aberto = true; break; }
  }
  return { aberto, hoje: desc };
}

/* -------------------- Configurações -------------------- */
function ConfigView({ toast }: any) {
  const [cfg, setCfg] = useState<any>(null);
  const [meta, setMeta] = useState('');
  const [teto, setTeto] = useState('');
  const [kill, setKill] = useState(false);
  const [plano, setPlano] = useState(297);
  const [saving, setSaving] = useState('');
  useEffect(() => {
    (async () => {
      const rows = await apiGet('painel-config');
      const c = rows[0]?.config;
      if (c && !c.erro) { setCfg(c); setMeta(String(c.meta_faturamento ?? '')); setTeto(String(c.teto_aprovacao_semanal ?? '')); setKill(!!c.kill_ativo); }
      else setCfg({});
      const v = Number(localStorage.getItem('rl-plano')); if (v > 0) setPlano(v);
    })();
  }, []);
  const save = async (chave: string, valor: string, label: string) => {
    setSaving(chave);
    const d = await apiPost('painel-config-set', { chave, valor });
    setSaving('');
    if (d.status === 'ok' || d.status === 'sem_mudanca') toast(`${label} salvo`);
    else toast('Sessão inválida', 'err');
  };
  const savePlano = (v: number) => { setPlano(v); localStorage.setItem('rl-plano', String(v || 0)); };
  const killToggle = async () => { const nv = !kill; setKill(nv); await save('kill_ativo', nv ? 'true' : 'false', nv ? 'Coleta pausada' : 'Coleta retomada'); };
  if (!cfg) return <div className="rl-grid"><div className="rl-skel" /><div className="rl-skel" /></div>;
  return (
    <div className="rl-gpage">
      <div className="rl-gsec-title">Metas & capacidade</div>
      <div className="rl-card">
        <div className="rl-field"><div className="rl-label">Meta de faturamento mensal (R$)</div>
          <div style={{ display: 'flex', gap: 8, marginTop: 6 }}>
            <input className="rl-input mono" inputMode="numeric" value={meta} onChange={(e) => setMeta(digits(e.target.value))} />
            <button className="rl-btn primary" disabled={saving === 'meta_faturamento'} onClick={() => save('meta_faturamento', meta || '0', 'Meta')}>Salvar</button>
          </div>
          <div className="rl-meta">Usada na barra de faturamento do Dashboard.</div>
        </div>
        <div className="rl-field"><div className="rl-label">Teto de aprovações por semana</div>
          <div style={{ display: 'flex', gap: 8, marginTop: 6 }}>
            <input className="rl-input mono" inputMode="numeric" value={teto} onChange={(e) => setTeto(digits(e.target.value))} />
            <button className="rl-btn primary" disabled={saving === 'teto_aprovacao_semanal'} onClick={() => save('teto_aprovacao_semanal', teto || '0', 'Teto')}>Salvar</button>
          </div>
          <div className="rl-meta">Limita quantos leads você aprova por semana — sua capacidade real de entrega.</div>
        </div>
        <div className="rl-field"><div className="rl-label">Valor do plano mensal (recorrência)</div>
          <div style={{ display: 'flex', gap: 8, marginTop: 6, alignItems: 'center' }}>
            <span className="mono" style={{ color: 'var(--sub)' }}>R$</span>
            <input className="rl-input mono" style={{ maxWidth: 180 }} inputMode="numeric" value={plano} onChange={(e) => savePlano(Number(digits(e.target.value)) || 0)} />
          </div>
          <div className="rl-meta">Base do cálculo de receita recorrente na aba Gestão. Salvo neste navegador.</div>
        </div>
      </div>

      <div className="rl-gsec-title" style={{ marginTop: 20 }}>Coleta automática</div>
      <div className="rl-card">
        <div style={{ display: 'flex', alignItems: 'center', justifyContent: 'space-between', gap: 12, flexWrap: 'wrap' }}>
          <div style={{ minWidth: 0 }}>
            <div className="rl-title" style={{ fontSize: 15 }}>{kill ? '⏸ Coleta pausada' : '● Coleta ativa'}</div>
            <div className="rl-meta" style={{ marginTop: 2 }}>Liga/desliga o robô que busca novas empresas (WF1) toda semana no Google Places.</div>
          </div>
          <button className={`rl-btn ${kill ? 'primary' : 'danger'}`} disabled={saving === 'kill_ativo'} onClick={killToggle}>{kill ? 'Retomar coleta' : 'Pausar coleta'}</button>
        </div>
      </div>
    </div>
  );
}

const dataBr = (s: any) => { if (!s) return ''; const d = new Date(s); if (isNaN(d.getTime())) return ''; return ('0' + d.getDate()).slice(-2) + '/' + ('0' + (d.getMonth() + 1)).slice(-2); };

function ClienteCard({ it, onPos }: any) {
  const cid = it.contrato_id;
  const ob = it.onboarding || {};
  const done = ONBOARDING.filter((s) => ob[s.key]).length;
  const pct = Math.round((done / ONBOARDING.length) * 100);
  const fone = digits(it.telefone);
  const nome = it.nome || 'Cliente';
  const [nota, setNota] = useState<string>(it.notas || '');
  const upStatus = it.upsell_status || 'pendente';
  const abrirWa = (msg: string) => { if (!fone) { alert('Cliente sem telefone cadastrado'); return; } window.open(`https://wa.me/55${fone}?text=${encodeURIComponent(msg)}`, '_blank'); };
  return (
    <div className="rl-card">
      <CardHead nome={nome} sub={it.fechado_em ? `fechado ${dataBr(it.fechado_em)}` : 'Cliente ativo'} note={it.valor ? brl(it.valor) : null} />

        <div className="rl-onbhead"><span className="rl-label" style={{ textTransform: 'none' }}>Entrega ({done}/{ONBOARDING.length})</span><span className="mono" style={{ fontSize: 12, color: 'var(--green)', fontWeight: 700 }}>{pct}%</span></div>

        <div className="rl-checks">
          {ONBOARDING.map((s) => {
            const on = !!ob[s.key];
            return (
              <button key={s.key} className={`rl-check${on ? ' on' : ''}`} onClick={() => onPos(cid, 'onboarding', { step: s.key, valor: String(!on) })}>

              </button>
            );
          })}

      </div>

      <div className="rl-actbox">
        <div className="rl-label" style={{ textTransform: 'none' }}>Avaliação 5⭐ + indicação</div>
        <div className="rl-actrow2">
